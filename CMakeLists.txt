cmake_minimum_required(VERSION 3.24)
project(BodySlots VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_INSTALL_RULES ON)
set(BODY_SLOTS_SKSE_SDK "" CACHE PATH "Public SKSE VR SDK root containing common/ and sksevr/")
set(BODY_SLOTS_HIGGS_SDK "" CACHE PATH "HIGGS synergy source checkout containing include/higgshandapi.h")
if(NOT EXISTS "${BODY_SLOTS_SKSE_SDK}/sksevr/skse64/PluginAPI.h" OR NOT EXISTS "${BODY_SLOTS_HIGGS_SDK}/include/higgshandapi.h")
    message(FATAL_ERROR "Set BODY_SLOTS_SKSE_SDK and BODY_SLOTS_HIGGS_SDK to the public SDK checkouts; see docs/game-adapter.md")
endif()
include("${BODY_SLOTS_HIGGS_SDK}/cmake/HiggsServices.cmake")
# Import only public contract headers. The HIGGS source include root also has
# RE/ names and must never shadow CommonLib's engine declarations.
set(BS_HIGGS_HEADERS "${CMAKE_CURRENT_BINARY_DIR}/contracts/higgs")
file(MAKE_DIRECTORY "${BS_HIGGS_HEADERS}")
foreach(header IN ITEMS higgsinterface001.h higgshandapi.h higgshandrouting.h higgsgrantmutation.h higgsinputcontext.h higgsinventory.h higgstransactions.h higgsnativeregions.h higgsauthorizeddispatch.h)
    configure_file("${BODY_SLOTS_HIGGS_SDK}/include/${header}" "${BS_HIGGS_HEADERS}/${header}" COPYONLY)
endforeach()
include(FetchContent)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.0.2 GIT_SHALLOW TRUE OVERRIDE_FIND_PACKAGE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG v1.15.0 GIT_SHALLOW TRUE OVERRIDE_FIND_PACKAGE)
FetchContent_Declare(rapidcsv GIT_REPOSITORY https://github.com/d99kris/rapidcsv.git GIT_TAG v8.99 GIT_SHALLOW TRUE SOURCE_SUBDIR src)
FetchContent_MakeAvailable(fmt spdlog rapidcsv)
set(RAPIDCSV_INCLUDE_DIRS "${rapidcsv_SOURCE_DIR}/src" CACHE PATH "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
# Consume the existing library. This task does not extend CommonLib.
FetchContent_Declare(CommonLibSSE GIT_REPOSITORY https://github.com/CharmedBaryon/CommonLibSSE-NG.git GIT_TAG v3.7.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(CommonLibSSE)
set(BS_CORE src/registry.cpp src/instances.cpp src/instance_configuration.cpp src/validation.cpp src/configuration_serialization.cpp src/native_placement.cpp src/native_policy.cpp src/package.cpp src/runtime.cpp src/runtime_saved.cpp src/zone_geometry.cpp src/manipulation.cpp src/content_policy.cpp src/main/interaction_regions.cpp)
add_library(BodySlotsGameAdapter OBJECT src/game_adapter.cpp src/main/higgs_native_regions.cpp src/main/higgs_regions_connection.cpp src/main/region_actions.cpp src/main/higgs_region_actions_connection.cpp)
target_include_directories(BodySlotsGameAdapter PRIVATE src/compat include src "${RAPIDCSV_INCLUDE_DIRS}" "${BS_HIGGS_HEADERS}")
target_link_libraries(BodySlotsGameAdapter PRIVATE CommonLibSSE::CommonLibSSE)
add_library(BodySlots SHARED ${BS_CORE} src/plugin.cpp $<TARGET_OBJECTS:BodySlotsGameAdapter>)
target_include_directories(BodySlots PRIVATE include src "${RAPIDCSV_INCLUDE_DIRS}" "${BODY_SLOTS_SKSE_SDK}" "${BODY_SLOTS_SKSE_SDK}/sksevr")
target_compile_definitions(BodySlots PRIVATE BODY_SLOTS_WITH_GAME_ADAPTER=1 WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(BodySlots PRIVATE CommonLibSSE::CommonLibSSE)
if(MSVC)
    target_compile_options(BodySlots PRIVATE /W4 /permissive- /EHsc /utf-8 /wd4200 /wd4201 /wd4324)
endif()
enable_testing()
add_executable(body_slots_runtime_tests tests/runtime_tests.cpp ${BS_CORE})
target_include_directories(body_slots_runtime_tests PRIVATE include src)
if(MSVC)
    target_compile_options(body_slots_runtime_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
    target_link_options(body_slots_runtime_tests PRIVATE /STACK:8388608)
